7.2
PercentDeadband
The
DataChangeFilter
in
OPC 10000-4
defines the conditions under which a data change notification shall be reported. This filter contains a
deadbandValue
which can be of type
AbsoluteDeadband
or
PercentDeadband
.
OPC 10000-4
already specifies the behaviour of the
AbsoluteDeadband
. The behaviour of the
PercentDeadband
type is specified as follows.
DeadbandType = PercentDeadband
For this type of deadband the
deadbandValue
is defined as the percentage of the
EURange
. That is, it applies only to
AnalogItems
with an
EURange
Property
that defines the typical value range for the item. This range shall be multiplied with the
deadbandValue
and then compared to the actual value change to determine the need for a data change notification. The following pseudo code shows how the deadband is calculated:
DataChange if (absolute value of (last cached value - current value) >              (deadbandValue/100.0) * ((high-low) of EURange)))
The range of the
deadbandValue
is from 0,0 to 100,0 per cent. Specifying a
deadbandValue
outside of this range will be rejected and reported with the
StatusCode
Bad_DeadbandFilterInvalid (see
Table 61
).
If the Value of the
MonitoredItem
is an array, then the deadband calculation logic shall be applied to each element of the array. If an element that requires a DataChange is found, then no further deadband checking is necessary and the entire array shall be returned.

    // ── ToC pane resizer ─────────────────────────────────────────────────────────
    // Writes --toc-w on <html> and persists it; the <head> script above replays it on
    // the next page, so the width survives normal server-rendered navigation.
    (function () {
        var DEFAULT = 300, MIN = 180, KEY = 'opc.tocWidth';
        var handle = document.getElementById('toc-resizer');
        var layout = document.querySelector('.spec-layout');
        if (!handle || !layout) return;

        // Never let the ToC squeeze the content pane below a readable width.
        function maxWidth() { return Math.max(MIN, Math.min(900, window.innerWidth - 320)); }

        function current() {
            var v = parseInt(getComputedStyle(document.documentElement)
                             .getPropertyValue('--toc-w'), 10);
            return isNaN(v) ? DEFAULT : v;
        }

        // The width the user asked for, before clamping — kept so a temporarily narrow
        // window does not permanently shrink their preference.
        var desired = current();

        function apply(px, persist) {
            var w = Math.round(Math.max(MIN, Math.min(maxWidth(), px)));
            document.documentElement.style.setProperty('--toc-w', w + 'px');
            handle.setAttribute('aria-valuenow', String(w));
            if (persist) {
                desired = w;
                try { localStorage.setItem(KEY, String(w)); } catch (e) { }
            }
            return w;
        }

        handle.setAttribute('aria-valuenow', String(desired));

        handle.addEventListener('pointerdown', function (e) {
            if (e.button !== 0) return;
            e.preventDefault();
            handle.setPointerCapture(e.pointerId);
            document.body.classList.add('toc-resizing');
            var originX = layout.getBoundingClientRect().left;

            function onMove(ev) { apply(ev.clientX - originX, false); }
            function onUp(ev) {
                handle.removeEventListener('pointermove', onMove);
                handle.removeEventListener('pointerup', onUp);
                handle.removeEventListener('pointercancel', onUp);
                document.body.classList.remove('toc-resizing');
                apply(ev.type === 'pointercancel' ? current() : ev.clientX - originX, true);
            }
            handle.addEventListener('pointermove', onMove);
            handle.addEventListener('pointerup', onUp);
            handle.addEventListener('pointercancel', onUp);
        });

        handle.addEventListener('dblclick', function () {
            apply(DEFAULT, true);
        });

        handle.addEventListener('keydown', function (e) {
            var step = e.shiftKey ? 50 : 10;
            if (e.key === 'ArrowLeft')       { apply(current() - step, true); }
            else if (e.key === 'ArrowRight') { apply(current() + step, true); }
            else if (e.key === 'Home')       { apply(DEFAULT, true); }
            else return;
            e.preventDefault();
        });

        // A narrowed window can invalidate the stored width; re-clamp the *preference*
        // (not the clamped value) so the chosen width returns when the window grows.
        // Below the breakpoint the ToC is an overlay drawer sized by the stylesheet
        // (min(--toc-w, 85vw)) — leave it alone rather than clamping against a viewport
        // the drawer does not share with the content.
        var narrow = window.matchMedia('(max-width: 768px)');
        function sync() { if (!narrow.matches) { apply(desired, false); } }
        window.addEventListener('resize', sync);
        sync();
    })();

    // ── Scroll active ToC item into view on load ─────────────────────────────────
    (function () {
        var pane   = document.getElementById('toc-pane');
        var active = pane && pane.querySelector('.toc-list a.active');
        if (active) active.scrollIntoView({ block: 'center' });
    })();

    // ── In-page find bar (Ctrl/Cmd+F) ────────────────────────────────────────────
    (function () {
        var bar   = document.getElementById('find-bar');
        var input = document.getElementById('find-input');
        var count = document.getElementById('find-count');
        var marks = [];
        var cur   = -1;
        var tocHits = [];   // ToC anchors holding a match — declipped while the find bar is open

        // Content first, then the ToC: reading matches come before navigational ones.
        function getPanes() {
            var content = document.getElementById('content') ||
                          document.querySelector('.content-pane') ||
                          document.querySelector('main');
            var toc = document.getElementById('toc-pane');
            return [content, toc].filter(Boolean);
        }

        function clearMarks() {
            marks.forEach(function (m) {
                if (!m.parentNode) return;
                m.parentNode.replaceChild(document.createTextNode(m.textContent), m);
            });
            getPanes().forEach(function (p) { p.normalize(); });
            tocHits.forEach(function (a) { a.classList.remove('find-hit'); });
            tocHits = [];
            marks = [];
            cur   = -1;
        }

        function highlightIn(pane, query) {
            var lq = query.toLowerCase();
            var walker = document.createTreeWalker(pane, NodeFilter.SHOW_TEXT, {
                acceptNode: function (node) {
                    var tag = node.parentElement && node.parentElement.tagName;
                    return (tag === 'SCRIPT' || tag === 'STYLE')
                        ? NodeFilter.FILTER_REJECT : NodeFilter.FILTER_ACCEPT;
                }
            });
            var nodes = [];
            var node;
            while ((node = walker.nextNode())) nodes.push(node);
            nodes.forEach(function (textNode) {
                var text = textNode.textContent;
                var lt   = text.toLowerCase();
                var pos  = 0;
                var frags = [];
                var found = false;
                while (pos < text.length) {
                    var idx = lt.indexOf(lq, pos);
                    if (idx < 0) { frags.push(document.createTextNode(text.slice(pos))); break; }
                    found = true;
                    if (idx > pos) frags.push(document.createTextNode(text.slice(pos, idx)));
                    var m = document.createElement('mark');
                    m.textContent = text.slice(idx, idx + query.length);
                    frags.push(m);
                    marks.push(m);
                    pos = idx + lq.length;
                }
                if (!found) return;
                var parent = textNode.parentNode;
                var ref    = textNode.nextSibling;
                // ToC titles are ellipsised to one line, so a match in the middle of a long
                // title would be invisible — declip the entry that holds it.
                var anchor = parent.closest ? parent.closest('.toc-list a') : null;
                if (anchor && !anchor.classList.contains('find-hit')) {
                    anchor.classList.add('find-hit');
                    tocHits.push(anchor);
                }
                parent.removeChild(textNode);
                frags.forEach(function (f) { parent.insertBefore(f, ref); });
            });
        }

        function highlight(query) {
            clearMarks();
            count.textContent = '';
            if (!query) return;
            getPanes().forEach(function (pane) { highlightIn(pane, query); });
            if (marks.length > 0) { scrollToMark(0); }
            else { count.textContent = 'No results'; }
        }

        function inToc(m) {
            return !!(m.closest && m.closest('#toc-pane'));
        }

        var tocOpenedByFind = false;
        function setDrawer(open) {
            document.body.classList.toggle('toc-open', open);
            var btn = document.querySelector('.toc-toggle');
            if (btn) btn.setAttribute('aria-expanded', String(open));
        }

        function scrollToMark(idx) {
            if (!marks.length) return;
            marks.forEach(function (m, i) { m.className = i === idx ? 'find-current' : ''; });
            var m = marks[idx];
            // On mobile the ToC is an off-screen drawer — open it before scrolling into it,
            // and put it back when the walk returns to the content pane.
            if (window.matchMedia('(max-width: 768px)').matches) {
                if (inToc(m)) { setDrawer(true); tocOpenedByFind = true; }
                else if (tocOpenedByFind) { setDrawer(false); tocOpenedByFind = false; }
            }
            m.scrollIntoView({ block: 'center', behavior: 'smooth' });
            cur = idx;
            count.textContent = (cur + 1) + ' of ' + marks.length + (inToc(m) ? ' · contents' : '');
        }

        function openBar() { bar.hidden = false; input.focus(); input.select(); }
        function closeBar() {
            clearMarks();
            if (tocOpenedByFind) { setDrawer(false); tocOpenedByFind = false; }
            bar.hidden = true; input.value = ''; count.textContent = '';
        }

        document.addEventListener('keydown', function (e) {
            if ((e.ctrlKey || e.metaKey) && e.key === 'f') { e.preventDefault(); openBar(); }
            if (e.key === 'Escape' && !bar.hidden) closeBar();
        });
        input.addEventListener('input', function () { highlight(input.value); });
        input.addEventListener('keydown', function (e) {
            if (e.key === 'Enter') {
                e.preventDefault();
                if (marks.length) scrollToMark(e.shiftKey
                    ? (cur - 1 + marks.length) % marks.length
                    : (cur + 1) % marks.length);
            }
        });
        document.getElementById('find-prev').addEventListener('click', function () {
            if (marks.length) scrollToMark((cur - 1 + marks.length) % marks.length);
        });
        document.getElementById('find-next').addEventListener('click', function () {
            if (marks.length) scrollToMark((cur + 1) % marks.length);
        });
        document.getElementById('find-close').addEventListener('click', closeBar);
    })();

    // ── Term click → keyword search ───────────────────────────────────────────────
    (function () {
        document.addEventListener('click', function (e) {
            var el = e.target.closest('em[data-term]');
            if (!el) return;
            e.preventDefault();
            var term = el.getAttribute('data-term');
            if (!term) return;
            var form   = document.querySelector('.header-search');
            var action = form ? form.getAttribute('action') : null;
            if (!action) {
                var m = window.location.pathname.match(/^\/specs\/([^/]+)/);
                action = m ? '/specs/' + m[1] + '/search' : null;
            }
            if (action) window.location.href = action + '?q=' + encodeURIComponent(term) + '&tab=term';
        });
    })();
